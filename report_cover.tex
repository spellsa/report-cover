\newcommand{\ReportCoverClearDate}[1]{%
  \@namedef{ReportCover#1Year}{}%
  \@namedef{ReportCover#1Month}{}%
  \@namedef{ReportCover#1Day}{}%
}

\def\ReportCoverTodayKeyword{today}

\newcommand{\ReportCoverSetTodayDate}[1]{%
  \@namedef{ReportCover#1Year}{\number\year}%
  \@namedef{ReportCover#1Month}{\number\month}%
  \@namedef{ReportCover#1Day}{\number\day}%
}

\newcommand{\ReportCoverSetDate}[2]{%
  \def\ReportCoverDateInput{#2}%
  \ifx\ReportCoverDateInput\@empty
    \ReportCoverClearDate{#1}%
  \else
    \ifx\ReportCoverDateInput\ReportCoverTodayKeyword
      \ReportCoverSetTodayDate{#1}%
    \else
      \expandafter\ReportCoverSetDateAux\expandafter{#1}#2\@nil
    \fi
  \fi
}

\def\ReportCoverSetDateAux#1#2 #3 #4\@nil{%
  \@namedef{ReportCover#1Year}{#2}%
  \@namedef{ReportCover#1Month}{#3}%
  \@namedef{ReportCover#1Day}{#4}%
}
